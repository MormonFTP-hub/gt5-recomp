cmake_minimum_required(VERSION 3.20)
project(gt5
    VERSION 0.1.0
    LANGUAGES C CXX
    DESCRIPTION "Gran Turismo 5 static recompilation"
)

set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(PS3RECOMP_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../ps3recomp" CACHE PATH "ps3recomp checkout")
get_filename_component(PS3RECOMP_DIR "${PS3RECOMP_DIR}" ABSOLUTE)

set(RECOMP_DIR "${CMAKE_CURRENT_SOURCE_DIR}/generated")
set(SPU_DIR "${CMAKE_CURRENT_SOURCE_DIR}/spu")

if(NOT EXISTS "${RECOMP_DIR}/ppu_recomp.h")
    message(FATAL_ERROR "No ppu_recomp.h in ${RECOMP_DIR}. Run scripts/relift.ps1 first.")
endif()

file(GLOB RECOMP_SOURCES CONFIGURE_DEPENDS "${RECOMP_DIR}/*.cpp")
list(LENGTH RECOMP_SOURCES RECOMP_SOURCE_COUNT)
message(STATUS "Lifted sources:  ${RECOMP_SOURCE_COUNT} file(s)")

# Movies decode through FFmpeg, loaded at run time; only its headers are needed to build.
set(PS3RECOMP_FFMPEG_RUNTIME ON CACHE BOOL "" FORCE)
set(GT5_FFMPEG_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../ffmpeg/ffmpeg-n9.0-latest-win64-lgpl-shared-9.0")
set(PS3RECOMP_FFMPEG_INCLUDE_DIR "${GT5_FFMPEG_DIR}/include" CACHE PATH "" FORCE)
add_subdirectory("${PS3RECOMP_DIR}" "${CMAKE_BINARY_DIR}/ps3recomp_sdk" EXCLUDE_FROM_ALL)

# The HLE NID table comes from the toolkit under build, regenerated by relift.ps1:
# one built against another revision can reference handlers this one lacks.
set(HLE_NID_TABLE "${CMAKE_CURRENT_SOURCE_DIR}/src/gen/ppu_hle_nids.cpp")

# The shared boot harness rather than a port main.cpp: it is the only entry point
# that turns on the live NV4097 -> D3D12 engine (RSX_LIVE_DRAW=1).
set(PS3RECOMP_SCAFFOLD
    "${PS3RECOMP_DIR}/runtime/ppu/tests/boot_main.cpp"
    "${PS3RECOMP_DIR}/runtime/ppu/ppu_loader.cpp"
    "${PS3RECOMP_DIR}/runtime/ppu/ppu_hle.cpp"
    "${PS3RECOMP_DIR}/runtime/ppu/ppu_fs.cpp"
    "${PS3RECOMP_DIR}/runtime/ppu/ppu_sysprx.cpp"
)

# Lifted SPU images, registered with the runtime by content fingerprint. Only the
# ones build_spu_workloads.py registered: the other images it lifted carry shuffle
# tables inside their code ranges and run in the SPU interpreter for now.
set(SPU_SOURCES "")
if(EXISTS "${SPU_DIR}/sources.cmake")
    include("${SPU_DIR}/sources.cmake")
    list(LENGTH SPU_GENERATED_SOURCES SPU_SOURCE_COUNT)
    message(STATUS "Lifted SPU:      ${SPU_SOURCE_COUNT} file(s)")
    # The musttail indirect-branch fast path has to be built here, by clang.
    list(APPEND SPU_SOURCES ${SPU_GENERATED_SOURCES} "${PS3RECOMP_DIR}/runtime/spu/spu_dispatch_mt.c")
endif()

# Raw SPU images captured from a run; see scripts/lift_spu_jobs.py.
file(GLOB SPU_JOB_SOURCES CONFIGURE_DEPENDS "${SPU_DIR}/jobs/*/spu_recomp.c")
if(SPU_JOB_SOURCES)
    list(LENGTH SPU_JOB_SOURCES SPU_JOB_COUNT)
    message(STATUS "Lifted SPU jobs: ${SPU_JOB_COUNT} image(s)")
    list(APPEND SPU_SOURCES ${SPU_JOB_SOURCES} "${CMAKE_CURRENT_SOURCE_DIR}/src/gen/spu_jobs.c")
endif()

add_executable(gt5
    src/gt5_longjmp.cpp
    src/gt5_debug.cpp
    src/gt5_widescreen.cpp
    ${PS3RECOMP_SCAFFOLD}
    ${HLE_NID_TABLE}
    ${RECOMP_SOURCES}
    ${SPU_SOURCES}
)

# Game-code mods (scripts/mods.py, MODDING.md): the start-up switch of each mod that applies to
# this EBOOT. The file only exists when one does.
file(GLOB GT5_MOD_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/gen/mods.cpp")
if(GT5_MOD_SOURCES)
    target_sources(gt5 PRIVATE ${GT5_MOD_SOURCES})
endif()

# Scoped so the huge PPU TUs keep their cached compile commands. /GS-: no stack-cookie check in
# every lifted SPU function (most are a basic block long; their locals hold no guest-sized buffers).
if(SPU_SOURCES)
    set_source_files_properties(${SPU_SOURCES} PROPERTIES
        INCLUDE_DIRECTORIES "${PS3RECOMP_DIR}/runtime/spu"
        COMPILE_DEFINITIONS SPU_USE_DISPATCH_MT
        COMPILE_OPTIONS "-O2;/w;/GS-;-mavx2;-mfma;-ffp-contract=off;-DSPU_X86_RZ_ALWAYS=1")
endif()
# GT5's sound job: its numbers are samples, and it was about 10% short of real time with a full grid.
# jD58063CD is the disc's; game update 2.17 replaces it with jCB9BD024 (spu_jobs_from_eboot.py, UPDATED).
foreach(_sound_job jD58063CD jCB9BD024)
    set_source_files_properties("${SPU_DIR}/jobs/${_sound_job}/spu_recomp.c" PROPERTIES
        COMPILE_OPTIONS "-O2;/w;/GS-;-mavx2;-mfma;-ffp-contract=off;-DSPU_X86_RZ_ALWAYS=1;-DSPU_X86_FAST_FLOAT=1")
endforeach()

target_include_directories(gt5 PRIVATE
    "${RECOMP_DIR}"
    "${PS3RECOMP_DIR}/include"
    "${PS3RECOMP_DIR}/runtime/ppu"
    "${PS3RECOMP_DIR}/runtime/spu"
    "${PS3RECOMP_DIR}/runtime/memory"
    "${PS3RECOMP_DIR}/runtime/syscalls"
    "${PS3RECOMP_DIR}/runtime/platform"
    "${PS3RECOMP_DIR}/libs/video"
    "${CMAKE_CURRENT_SOURCE_DIR}"
)

target_link_libraries(gt5 PRIVATE ps3recomp_runtime dbghelp user32 gdi32 winmm)
# Line numbers for the runtime too, so a profile names the RSX thread's hot lines (used to be a hand edit
# of build.ninja that every CMake regeneration lost).
target_compile_options(ps3recomp_runtime PRIVATE /Z7)
# The DLLs the runtime opens sit next to the exe.
add_custom_command(TARGET gt5 POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "${GT5_FFMPEG_DIR}/bin/avcodec-63.dll" "${GT5_FFMPEG_DIR}/bin/avutil-61.dll" "${GT5_FFMPEG_DIR}/bin/swresample-7.dll"
    "$<TARGET_FILE_DIR:gt5>")
set_target_properties(gt5 PROPERTIES LINKER_LANGUAGE CXX)

target_compile_options(gt5 PRIVATE /W3 /wd4100 /wd4244 /wd4267 /Zc:__cplusplus -fno-strict-aliasing -fwrapv)
target_compile_definitions(gt5 PRIVATE _CRT_SECURE_NO_WARNINGS)

# Machine-generated: warnings off, and /bigobj because one TU passes the default
# object format's section limit. -O1 keeps these TUs' build time and memory sane;
# the last -O on the line wins, so this overrides Release's /O2 for them alone.
# -O2 with this CPU's instructions (2026-11-05): the largest chunk builds in 71 s against 67 s at -O1, under 1 GB.
# MOVBE folds the byte swap of every big-endian guest load and store into the access itself.
# No FMA contraction, so floating-point results match the -O1 build. -mfma lets ppu_vmx.h do
# vmaddfp/vnmsubfp four lanes at once with the same results (every AVX2 CPU has FMA; contraction
# stays off, so nothing else changes).
set_source_files_properties(${RECOMP_SOURCES} PROPERTIES COMPILE_OPTIONS "/w;/bigobj;-O2;-mavx2;-mfma;-mbmi;-mbmi2;-mmovbe;-mlzcnt;-mpopcnt;-ffp-contract=off")

# Keep symbols so crash RVAs can be symbolized.
target_compile_options(gt5 PRIVATE /Z7)
target_link_options(gt5 PRIVATE /DEBUG /INCREMENTAL:NO)
